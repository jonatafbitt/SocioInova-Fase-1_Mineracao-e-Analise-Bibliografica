@echo off
TITLE Observatorio de Inovacao - IFBA/UFBA
CLS

echo ======================================================
echo    INICIANDO AMBIENTE DE PESQUISA DOUTORAL
echo ======================================================
echo.

:: 1. Ativa o ambiente virtual
echo [1/3] Ativando Ambiente Virtual (venv)...
call .\venv\Scripts\activate
if %errorlevel% neq 0 (
    echo.
    echo [ERRO] Nao foi possivel ativar o venv! 
    echo Verifique se a pasta 'venv' existe.
    pause
    exit
)

:: 2. Executa o Script de Saude do Sistema
echo [2/3] Rodando Diagnostico de Saude...
python verificar_sistema.py
if %errorlevel% neq 0 (
    echo.
    echo [AVISO] Foram encontrados problemas no diagnostico acima.
    echo Deseja continuar mesmo assim? (S/N)
    set /p choice=
    if /i "%choice%" neq "S" exit
)

:: 3. Oferece o Menu de Opcoes
:menu
echo.
echo ======================================================
echo [3/3] SELECIONE A FERRAMENTA DESEJADA:
echo ======================================================
echo 1. Abrir Dashboard (Visualizacao e IA)
echo 2. Rodar Minerador (OpenAlex)
echo 3. Sincronizar Tudo (Google Drive)
echo 4. Sair
echo ======================================================
set /p op=Escolha uma opcao: 

if "%op%"=="1" goto dashboard
if "%op%"=="2" goto mineração
if "%op%"=="3" goto sincronizar
if "%op%"=="4" exit

:dashboard
echo.
echo Iniciando o Dashboard...
streamlit run app_principal.py
pause
goto menu

:mineração
echo.
echo Iniciando Coleta de Dados...
python scripts/minerador_openalex.py
pause
goto menu

:sincronizar
echo.
echo Iniciando Fluxo Mestre de Sincronizacao...
python scripts/execucao_mestre.py
pause
goto menu
