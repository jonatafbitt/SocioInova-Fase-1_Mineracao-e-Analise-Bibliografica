import streamlit as st
import pandas as pd
import os
import ollama
from langchain_community.document_loaders import PyPDFLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain_community.embeddings import OllamaEmbeddings

# --- CONFIGURAÇÕES DE CAMINHOS ---
PASTA_DATA = "data/producoes_mineradas.csv"
PASTA_PDFS = "meus_pdfs"
DB_DIR = "data/vector_db"

# Configuração da Página
st.set_page_config(page_title="Observatório de Inovação - IFBA", layout="wide")

# --- INTERFACE LATERAL ---
st.sidebar.title("🔬 Gestão de Pesquisa")
modelo_ia = st.sidebar.selectbox(
    "Cérebro da IA (LLM):", 
    ["phi3", "llama3", "mistral"], 
    help="Phi-3: Muito mais rápido para processamento de textos e indexação."
)

st.sidebar.divider()
if st.sidebar.button("☁️ Sincronizar com Google Drive"):
    st.sidebar.info("Executando scripts/execucao_mestre.py...")
    os.system("python scripts/execucao_mestre.py")
    st.sidebar.success("Sincronização Concluída!")

# --- CORPO PRINCIPAL ---
st.title("🎓 Observatório de Sociologia da Inovação")
aba_web, aba_local = st.tabs(["🌐 Pesquisa OpenAlex", "📚 Minha Biblioteca (PDFs)"])

# --- ABA 1: PESQUISA WEB (OPENALEX / GRAFICOS) ---
with aba_web:
    if not os.path.exists(PASTA_DATA):
        st.error("Base de dados não encontrada. Rode o minerador primeiro.")
    else:
        df = pd.read_csv(PASTA_DATA)
        
        # Mapeamento dinâmico
        colunas_possiveis = {
            'ano': ['ano', 'publication_year', 'year'],
            'titulo': ['titulo', 'title', 'display_name'],
            'autores': ['autores', 'authors', 'author_names'],
            'resumo': ['resumo', 'abstract', 'description'],
            'idioma': ['idioma', 'language']
        }
        
        col_enc = {k: next((c for c in v if c in df.columns), None) for k, v in colunas_possiveis.items()}
        
        # --- SEÇÃO DE GRÁFICOS (NOVIDADE) ---
        st.subheader("📊 Panorama da Produção Científica")
        g1, g2 = st.columns(2)
        
        with g1:
            if col_enc['ano']:
                st.write("**Publicações por Ano**")
                contagem_ano = df[col_enc['ano']].value_counts().sort_index()
                st.bar_chart(contagem_ano)
        
        with g2:
            if col_enc['idioma']:
                st.write("**Distribuição por Idioma**")
                contagem_idioma = df[col_enc['idioma']].value_counts()
                st.bar_chart(contagem_idioma)

        st.divider()
        
        # Tabela e Filtros
        col_tit = col_enc['titulo'] if col_enc['titulo'] else df.columns[0]
        busca = st.text_input(f"🔍 Filtrar obras por {col_tit}:")
        df_filtrado = df[df[col_tit].str.contains(busca, case=False, na=False)] if busca else df
        
        col_visiveis = [v for k, v in col_enc.items() if v and k != 'resumo']
        st.dataframe(df_filtrado[col_visiveis].head(100), use_container_width=True)

        # Inteligência Analítica
        st.header("🤖 Inteligência Analítica")
        selecao = st.selectbox("Selecione uma obra para análise profunda:", df_filtrado[col_tit].tolist())
        
        if selecao:
            art = df_filtrado[df_filtrado[col_tit] == selecao].iloc[0]
            if col_enc['resumo'] and pd.notna(art[col_enc['resumo']]):
                texto = art[col_enc['resumo']]
                c1, c2 = st.columns(2)
                with c1:
                    if st.button("🚀 Gerar Fichamento Ágil"):
                        with st.spinner("Analisando..."):
                            prompt = f"Realize o fichamento estruturado (Objetivo, Metodologia, Achados) deste resumo sociológico: {texto}"
                            resp = ollama.chat(model=modelo_ia, messages=[{'role': 'user', 'content': prompt}])
                            st.info(f"Análise via {modelo_ia.upper()}")
                            st.markdown(resp['message']['content'])
                with c2:
                    if st.button("🔍 Analisar Tendência"):
                        with st.spinner("Avaliando viés..."):
                            prompt = f"Analise se o resumo tem viés OTIMISTA TECNOLÓGICO ou CRÍTICO-REFLEXIVO: {texto}"
                            resp = ollama.chat(model=modelo_ia, messages=[{'role': 'user', 'content': prompt}])
                            st.warning(f"Avaliação via {modelo_ia.upper()}")
                            st.markdown(resp['message']['content'])

# --- ABA 2: MINHA BIBLIOTECA (PDFs LOCAIS) ---
with aba_local:
    st.header("📚 Assistente de Leitura de PDFs")
    
    # --- LISTAGEM PRÉVIA DE ARQUIVOS (NOVIDADE) ---
    if os.path.exists(PASTA_PDFS):
        arquivos_pdf = [f for f in os.listdir(PASTA_PDFS) if f.endswith(".pdf")]
        if arquivos_pdf:
            with st.expander(f"📂 Ver {len(arquivos_pdf)} PDFs detectados na pasta"):
                for a in arquivos_pdf:
                    st.write(f"- {a}")
        else:
            st.warning("Nenhum PDF encontrado na pasta 'meus_pdfs'.")
    
    st.divider()
    
    if st.button("🔄 Escanear e Indexar para IA"):
        if not arquivos_pdf:
            st.error("Sem arquivos para indexar.")
        else:
            barra = st.progress(0)
            status = st.empty()
            documentos = []
            
            for i, arq in enumerate(arquivos_pdf):
                status.markdown(f"📖 Lendo: `{arq}`")
                barra.progress((i + 1) / len(arquivos_pdf))
                try:
                    loader = PyPDFLoader(os.path.join(PASTA_PDFS, arq))
                    documentos.extend(loader.load())
                except Exception as e:
                    st.warning(f"Erro em {arq}: {e}")
            
            status.markdown("✂️ Fragmentando textos...")
            text_splitter = RecursiveCharacterTextSplitter(chunk_size=2000, chunk_overlap=200)
            splits = text_splitter.split_documents(documentos)
            
            status.markdown(f"🧠 Gerando vetores com {modelo_ia}...")
            Chroma.from_documents(documents=splits, embedding=OllamaEmbeddings(model=modelo_ia), persist_directory=DB_DIR)
            
            barra.empty()
            status.empty()
            st.success(f"✅ Concluído! {len(splits)} trechos prontos para conversa.")

    st.divider()
    pergunta = st.text_input("💬 Pergunte algo à sua biblioteca:")
    if pergunta:
        if not os.path.exists(DB_DIR):
            st.error("Indexe os PDFs primeiro.")
        else:
            with st.spinner("Buscando fontes..."):
                vector_db = Chroma(persist_directory=DB_DIR, embedding_function=OllamaEmbeddings(model=modelo_ia))
                docs = vector_db.similarity_search(pergunta, k=3)
                contexto = "\n\n".join([d.page_content for d in docs])
                prompt_rag = f"Baseado no contexto: {contexto}. Responda: {pergunta}"
                final_resp = ollama.chat(model="llama3", messages=[{'role': 'user', 'content': prompt_rag}])
                st.markdown("### 🤖 Resposta da IA:")
                st.markdown(final_resp['message']['content'])
                with st.expander("Ver fontes"):
                    for d in docs:
                        st.write(f"**Fonte:** {d.metadata.get('source')}")
                        st.write(d.page_content[:300] + "...")
