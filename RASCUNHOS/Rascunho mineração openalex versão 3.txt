import requests
import pandas as pd
import os

# ======================================================
# CONFIGURAÇÃO DE TERMOS MULTILINGUES (VERSÃO AMPLIADA)
# ======================================================

# 1. CERNE OBRIGATÓRIO (O artigo DEVE ter um destes termos de núcleo)
cerne = [
    '"Sociology of Innovation"', '"Sociologia da Inovação"', 
    '"Sociologie de l\'innovation"', '"Innovationssoziologie"',
    '"Federal Institute"', '"Institutos Federais"', '"Rede Federal"'
]

# 2. TEMAS COMPLEMENTARES (O artigo deve conter ao menos um destes para refinar a busca)
complementos = [
    '"Social Innovation"', '"Inovação Social"', '"Innovation Sociale"',
    '"Decolonial"', '"Décoloniale"', '"Post-colonial"',
    '"Brazil"', '"Brasil"', '"Brésil"'
]

# Montagem da Query: (Cerne A OR Cerne B...) AND (Complemento 1 OR Complemento 2...)
query_cerne = f"({' OR '.join(cerne)})"
query_complementos = f"({' OR '.join(complementos)})"

QUERY_FINAL = f"{query_cerne} AND {query_complementos}"
LIMITE_RESULTADOS = 200 
# ======================================================

def reconstruir_abstract(inverted_index):
    """Reconstroi o resumo que o OpenAlex entrega em formato de índice invertido."""
    if not inverted_index:
        return "Resumo não disponível."
    try:
        max_pos = max([pos for positions in inverted_index.values() for pos in positions])
        texto_reconstruido = [None] * (max_pos + 1)
        for palavra, posicoes in inverted_index.items():
            for pos in posicoes:
                texto_reconstruido[pos] = palavra
        return ' '.join([palavra for palavra in texto_reconstruido if palavra is not None])
    except Exception:
        return "Erro ao processar resumo."

def executar_mineracao(termo=QUERY_FINAL, limite=LIMITE_RESULTADOS):
    url = "https://api.openalex.org/works"
    params = {
        'search': termo,
        'per_page': limite,
        'sort': 'cited_by_count:desc'
    }
    
    print(f"🛰️ Iniciando mineração multilingue no OpenAlex...")
    print(f"🔍 Query Otimizada: {termo[:120]}...") 
    
    try:
        response = requests.get(url, params=params, timeout=30)
        response.raise_for_status()
        
        results = response.json().get('results', [])
        lista_obras = []
        
        for obra in results:
            resumo = reconstruir_abstract(obra.get('abstract_inverted_index'))
            
            # Extração segura de autores
            autores_lista = [a.get('author', {}).get('display_name') for a in obra.get('authorships', [])]
            autores_str = ", ".join([a for a in autores_lista if a])

            lista_obras.append({
                'id': obra.get('id'),
                'titulo': obra.get('display_name'),
                'ano': obra.get('publication_year'),
                'autores': autores_str,
                'citacoes': obra.get('cited_by_count'),
                'doi': obra.get('doi'),
                'resumo': resumo,
                'idioma': obra.get('language')
            })
        
        if not lista_obras:
            print("⚠️ Nenhum resultado encontrado. A query ainda pode estar muito restrita.")
            return None

        # Salvar em CSV para o Dashboard
        df = pd.DataFrame(lista_obras)
        caminho_csv = os.path.join('data', 'producoes_mineradas.csv')
        df.to_csv(caminho_csv, index=False, encoding='utf-8')
        
        print(f"✅ SUCESSO: {len(df)} artigos (PT/EN/FR/DE) salvos em {caminho_csv}")
        return df

    except requests.exceptions.RequestException as e:
        print(f"❌ Erro na API: {e}")
        return None

if __name__ == "__main__":
    # Garante que a pasta data existe
    if not os.path.exists('data'):
        os.makedirs('data')
    
    executar_mineracao()
