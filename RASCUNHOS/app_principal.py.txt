import streamlit as st
import pandas as pd
import plotly.express as px
import os

# Configuração visual da página
st.set_page_config(page_title="Dashboard de Pesquisa - IFBA", layout="wide")

st.title("🔬 Observatório de Sociologia da Inovação")
st.markdown("Análise de produções científicas para suporte à tese doutoral (UFBA).")

# Caminho do arquivo de dados
caminho_csv = os.path.join('data', 'producoes_mineradas.csv')

# Verificar se os dados existem
if os.path.exists(caminho_csv):
    df = pd.read_csv(caminho_csv)

    # Barra Lateral para Filtros
    st.sidebar.header("Filtros de Pesquisa")
    ano_min = int(df['ano'].min())
    ano_max = int(df['ano'].max())
    anos_selecionados = st.sidebar.slider("Intervalo de Anos", ano_min, ano_max, (ano_min, ano_max))

    # Filtrar o DataFrame
    df_filtrado = df[(df['ano'] >= anos_selecionados[0]) & (df['ano'] <= anos_selecionados[1])]

    # Métricas Principais
    col1, col2, col3 = st.columns(3)
    col1.metric("Total de Artigos", len(df_filtrado))
    col2.metric("Artigo mais Citado", f"{df_filtrado['citacoes'].max()} citações")
    col3.metric("Ano Inicial", anos_selecionados[0])

    # Visualizações Gráficas
    st.divider()
    col_graf1, col_graf2 = st.columns(2)

    with col_graf1:
        st.subheader("📈 Evolução Temporal")
        fig_ano = px.histogram(df_filtrado, x="ano", nbins=20, labels={'ano': 'Ano', 'count': 'Qtd'}, 
                               title="Produção por Ano", color_discrete_sequence=['#0068c9'])
        st.plotly_chart(fig_ano, use_container_width=True)

    with col_graf2:
        st.subheader("📊 Top Artigos por Citação")
        top_citados = df_filtrado.nlargest(10, 'citacoes')
        fig_cit = px.bar(top_citados, x='citacoes', y='titulo', orientation='h', 
                         title="Impacto das Obras", labels={'citacoes': 'Citações', 'titulo': 'Obra'})
        # Inverter para o maior ficar em cima
        fig_cit.update_layout(yaxis={'categoryorder':'total ascending'})
        st.plotly_chart(fig_cit, use_container_width=True)

    # Tabela Detalhada
    st.divider()
    st.subheader("📋 Lista Detalhada de Produções")
    st.dataframe(df_filtrado[['titulo', 'ano', 'citacoes', 'doi']], use_container_width=True)

else:
    st.warning("⚠️ O arquivo de dados ainda não foi gerado. Por favor, execute o script de mineração primeiro.")
